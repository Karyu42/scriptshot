  -- Seconds between dashes. Leave it out to use the game's (1.5 s).

  -- Runs when you press Dash and the cooldown is over.
  -- dir is the way you are moving ({x, y}, length 1), or where you aim when
  -- standing still. Return false to say "no dash happened".
  on_dash = function(ctx, dir)
    -- Abilities go here, before or after the dash. For example, once the

    -- The built-in dash. Only player:dash() gives the 0.2 s invulnerability,

    return player:dash(math.atan(dir.y, dir.x))
  end,
}
